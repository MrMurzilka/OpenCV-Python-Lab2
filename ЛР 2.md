## 3. Теоретическая часть

### 3.1. Наследование в Python

**Наследование** — механизм, позволяющий создавать новые классы на основе существующих, наследуя их атрибуты и методы.

#### 3.1.1. Простое наследование

```python
class MediaFile:
    """Базовый класс для медиафайлов."""
    
    def __init__(self, filename):
        self.filename = filename
    
    def get_filename(self):
        return self.filename


class ImageFile(MediaFile):
    """Класс для изображений."""
    
    def __init__(self, filename, width, height):
        super().__init__(filename)
        self.width = width
        self.height = height
    
    def get_dimensions(self):
        return f"{self.width}x{self.height}"
```

```python
class CompressibleMedia(MediaFile):
```

### 3.2. Абстрактные классы и интерфейсы

**Абстрактный класс** — класс, который нельзя создать напрямую, но от которого можно наследоваться.

#### 3.2.1. Использование ABC (Abstract Base Classes)

```python
from abc import ABC, abstractmethod


class MediaProcessor(ABC):
    """Абстрактный класс для обработчиков медиафайлов."""
    
    @abstractmethod
    def process(self, file_path):
        """Обрабатывает медиафайл."""
        pass
    
    @abstractmethod
    def validate(self, file_path):
        """Валидирует медиафайл."""
        pass


class ImageProcessor(MediaProcessor):
    """Конкретный обработчик изображений."""
    
    def process(self, file_path):
        # Реализация обработки изображения
        print(f"Processing image: {file_path}")
        return True
    
    def validate(self, file_path):
        # Валидация изображения
        import os
        return os.path.exists(file_path)
```

#### 3.2.2. Пример использования абстрактных классов

### 3.3. Обработка исключений

**Исключения** — механизм обработки ошибок во время выполнения программы.

#### 3.3.1. Стандартные исключения Python

```python
try:
```

```python
class MediaError(Exception):
```

### 3.4. Практическое применение

#### 3.4.1. Класс для загрузки медиафайлов с обработкой ошибок

```python
class MediaLoader:
    """Класс для загрузки медиафайлов."""
    
    def load_image(self, file_path):
        """Загружает изображение с обработкой ошибок."""
        try:
            import cv2
            img = cv2.imread(file_path)
            if img is None:
                raise FileLoadError(file_path, "Файл не является изображением")
            return img
        except ImportError:
            raise MediaError("OpenCV не установлен", file_path)
        except Exception as e:
            raise FileLoadError(file_path, str(e))
    
    def load_video(self, file_path):
        """Загружает видео с обработкой ошибок."""
        try:
            import cv2
            cap = cv2.VideoCapture(file_path)
            if not cap.isOpened():
                raise FileLoadError(file_path, "Не удалось открыть видео")
            return cap
        except Exception as e:
            raise MediaError(f"Ошибка загрузки видео: {e}", file_path)
```

#### 3.4.2. Использование обработчика ошибок

```python
import cv2

def process_media_file(loader, file_path):
    """Обрабатывает медиафайл с полной обработкой ошибок."""
    try:
        # Определяем тип файла
        if file_path.lower().endswith(('.jpg', '.png', '.jpeg')):
            img = loader.load_image(file_path)
            print(f"Изображение загружено: {img.shape}")
        elif file_path.lower().endswith(('.mp4', '.avi', '.mkv')):
            cap = loader.load_video(file_path)
            fps = cap.get(cv2.CAP_PROP_FPS)
            print(f"Видео загружено: {fps} FPS")
            cap.release()
        else:
            raise MediaError(f"Неизвестный тип файла: {file_path}", file_path)
    except FileLoadError as e:
        print(f"Ошибка загрузки: {e}")
    except InvalidDataError as e:
        print(f"Ошибка данных: {e}")
    except MediaError as e:
        print(f"Общая ошибка: {e}")
```

### 3.5. Конкретные реализации фильтров как наследники абстрактного интерфейса

Продолжим пример с ЛР №1, реализуя конкретные фильтры как наследников абстрактного класса:

```python
from abc import ABC, abstractmethod


class ImageFilter(ABC):
    """Абстрактный базовый класс для фильтров изображений."""
    
    @abstractmethod
    def apply(self, image_data):
        """Применяет фильтр к изображению."""
        pass


class BrightnessFilter(ImageFilter):
    """Фильтр для изменения яркости изображения."""
    
    def __init__(self, factor: float = 1.5):
        self.factor = factor
    
    def apply(self, image_data):
        """Увеличивает/уменьшает яркость."""
        # ВАШ КОД: реализуйте изменение яркости с обработкой ошибок
        try:
            import cv2
            import numpy as np
            result = cv2.convertScaleAbs(image_data, alpha=self.factor, beta=0)
            return result
        except Exception as e:
            raise InvalidDataError("image", "Ошибка при изменении яркости")


class ContrastFilter(ImageFilter):
    """Фильтр для изменения контраста изображения."""
    
    def __init__(self, factor: float = 1.5):
        self.factor = factor
    
    def apply(self, image_data):
        """Увеличивает/уменьшает контраст."""
        # ВАШ КОД: реализуйте изменение контраста с обработкой ошибок
        try:
            import cv2
            import numpy as np
            mean = 128
            result = (image_data.astype(float) - mean) * self.factor + mean
            return np.clip(result, 0, 255).astype(np.uint8)
        except Exception as e:
            raise InvalidDataError("image", "Ошибка при изменении контраста")


class BlurFilter(ImageFilter):
    """Фильтр для размытия изображения."""
    
    def __init__(self, kernel_size: int = 5):
        self.kernel_size = kernel_size
    
    def apply(self, image_data):
        """Применяет размытие."""
        # ВАШ КОД: реализуйте размытие с обработкой ошибок
        try:
            import cv2
            result = cv2.GaussianBlur(image_data, (self.kernel_size, self.kernel_size), 0)
            return result
        except Exception as e:
            raise InvalidDataError("image", "Ошибка при размытии")
```

**Пример использования:**

```python
def apply_filters_to_image(image_path, filters):
    """Применяет последовательность фильтров к изображению."""
    try:
        import cv2
        img = cv2.imread(image_path)
        if img is None:
            raise FileLoadError(image_path, "Не удалось загрузить изображение")
        
        result = img
        for filter_obj in filters:
            result = filter_obj.apply(result)
        
        return result
    except MediaError as e:
        print(f"Ошибка обработки: {e}")
```

## 6. Обработка ошибок

При работе с мультимедийными файлами необходимо обрабатывать следующие ошибки:

1. **Файл не найден** — проверьте существование файла
2. **Неверный формат** — убедитесь, что файл имеет правильный формат
3. **Повреждённый файл** — обработайте повреждённые медиафайлы
4. **Отсутствуют зависимости** — проверьте наличие установленных библиотек

Пример обработки ошибок:

```python
def safe_process_media(processor, file_path):
    """Безопасная обработка медиафайла."""
    try:
        if not processor.validate(file_path):
            raise InvalidDataError(file_path)
        
        result = processor.process(file_path)
        return result
        
    except FileLoadError as e:
        print(f"Ошибка загрузки: {e}")
        return None
    except InvalidDataError as e:
        print(f"Ошибка данных: {e}")
        return None
    except MediaError as e:
        print(f"Ошибка обработки: {e}")
        return None
```
